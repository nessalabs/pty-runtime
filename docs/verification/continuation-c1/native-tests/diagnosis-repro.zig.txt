test "TEMP seed509 continuation diagnosis" {
    const S = Stream(ContinuationTestHandler);
    var stream = S.init(.{
        .allocator = testing.allocator,
        .handler = .{},
        .continuation_max_bytes = 64 * 1024,
    });
    defer stream.deinit();

    const feeds = [_][]const u8{
        &.{ 0x1b, 0x5b, 0x48, 0x1b, 0x5b, 0x32, 0x4a },
        &.{ 0xe2, 0x82 },
        &.{ 0x1b, 0x5b, 0x33, 0x38, 0x3b, 0x32, 0x3b, 0x31, 0x30, 0x3b },
        &.{
            0xa6, 0xaf, 0xf2, 0x76, 0x08, 0xc3, 0xb2, 0xd7, 0xc1, 0xb4,
            0xbb, 0x22, 0x66, 0x6f, 0x6f, 0xaa, 0x1b, 0xbd, 0x58, 0xc4,
            0x2d, 0xbb, 0x6f, 0x14, 0x1c, 0xd7, 0xe9, 0x90, 0x6a, 0x6b,
            0xdc, 0xcf, 0x28, 0xd9, 0xf4, 0xbb, 0x9f, 0x59, 0xdf, 0x3c,
            0x6e, 0xbf, 0xaf, 0x4b, 0xf7, 0x0e, 0xad, 0x4c, 0x63, 0xcd,
            0xaf, 0x57, 0xa0, 0x35, 0xe6, 0xdd, 0xec, 0x30, 0xdb, 0xc6,
            0x22, 0x2e, 0xca, 0xaa, 0x5c, 0x82, 0x3b, 0xb9, 0xbc, 0x4f,
            0x6d, 0x36, 0xfd, 0x5b, 0x2c, 0x65, 0x91, 0x58, 0x4d, 0x4c,
            0x09, 0x20, 0x2f, 0xd0, 0xbf, 0x50, 0xc7,
        },
    };
    for (feeds, 0..) |feed, i| {
        stream.nextSlice(feed);
        std.debug.print(
            "feed {d}: parser={s} utf8={d} retained={d}\n",
            .{
                i,
                @tagName(stream.parser.state),
                stream.utf8decoder.state,
                stream.continuation.?.bytes.items.len,
            },
        );
    }

    var buf: [4096]u8 = undefined;
    var writer: std.Io.Writer = .fixed(&buf);
    try stream.writeContinuation(&writer);
    std.debug.print("written={x}\n", .{writer.buffered()});
    std.debug.print(
        "validate={any}\n",
        .{continuationpkg.validate(writer.buffered())},
    );
}
